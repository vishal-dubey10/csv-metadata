# CSVForge — production dependencies for the backend Lambda.
#
# WHY EXACT VERSIONS (==):
#   These deps are pinned to exact versions (== not >=) deliberately.
#   A dependency like `fastapi>=0.100` could silently upgrade to a version
#   with a breaking change the next time you run `pip install` on a fresh
#   machine or in CI.  Exact pins make builds reproducible: the same versions
#   install every time, everywhere.  When you want to upgrade, do it explicitly
#   and run the full test suite to catch breakage.
#
# HOW TO UPDATE A PACKAGE:
#   1. Change the version here.
#   2. Run: pip install -r requirements.txt
#   3. Run: PYTHONPATH=. pytest tests/unit/ -v   (all 53 must pass)
#   4. Commit the updated requirements.txt.

# FastAPI — the web framework that powers the CSVForge API.
# FastAPI reads Python type annotations to auto-validate request/response
# bodies, auto-generate OpenAPI docs (/docs), and inject dependencies (Depends).
# Deployed to AWS Lambda via Mangum (see below).
fastapi==0.141.1

# Uvicorn — ASGI server used for LOCAL development only.
# In production, Lambda calls the ASGI app directly through Mangum; uvicorn is
# not needed there. Locally, uvicorn runs the app so you can hit it at
# http://localhost:8000. The [standard] extra includes watchfiles (hot reload),
# websockets support, and faster event loop bindings.
uvicorn[standard]==0.51.0

# Pydantic — data validation and settings management.
# FastAPI uses Pydantic v2 to validate incoming request JSON, coerce types,
# and serialize response objects. Every FastAPI request/response schema is a
# Pydantic BaseModel. Also used by pydantic-settings to read .env files and
# environment variables with type safety.
pydantic==2.13.4

# SQLAlchemy — the ORM (Object-Relational Mapper) and SQL toolkit.
# Maps Python classes (Job, etc.) to PostgreSQL tables and provides a
# high-level query API. Version 2.0 introduced typed mapped columns
# (Mapped[str], mapped_column()) that work with mypy/pyright.
# The ORM is used exclusively; SQLAlchemy Core is not used directly.
sqlalchemy==2.0.54

# Alembic — database schema migration tool built on top of SQLAlchemy.
# Manages the evolution of the PostgreSQL schema over time: each migration
# script represents one schema change (create table, add column, add index).
# Without Alembic, developers would need to run raw SQL ALTER TABLE statements
# manually and coordinate across environments. Alembic keeps an alembic_version
# table in the database to track which migrations have been applied.
alembic==1.20.0

# psycopg[binary] — PostgreSQL database driver (psycopg 3, the modern version).
# SQLAlchemy needs a driver to actually talk to PostgreSQL. psycopg 3 (imported
# as `psycopg`, not `psycopg2`) is the current generation: async-capable,
# faster, and actively maintained. The [binary] extra bundles the compiled C
# extension as a pre-built wheel so there's no need for libpq-dev or a C
# compiler in the Lambda build environment.
# Note: the URL prefix is `postgresql+psycopg://` (not `postgresql+psycopg2://`).
psycopg[binary]==3.3.3

# boto3 — the AWS SDK for Python.
# Used in two places: (1) backend/db/connection.py calls Secrets Manager to
# fetch DB credentials in production; (2) main.py will call S3 (presigned URLs)
# and SQS (job enqueuing) in Day 2+. boto3 bundles botocore (the low-level HTTP
# layer) and abstractions for every AWS service.
boto3==1.43.63

# Mangum — ASGI adapter for AWS Lambda + API Gateway.
# Lambda's event handler protocol (event, context) does not speak ASGI.
# Mangum translates an API Gateway HTTP event into an ASGI request and
# passes it to the FastAPI app, then translates the ASGI response back into
# the format API Gateway expects. Without Mangum, the entire FastAPI app would
# need to be rewritten to use Lambda's raw event/context model.
mangum==0.22.0

# python-dotenv — reads key=value pairs from a .env file into os.environ.
# Used in connection.py (load_dotenv()) so that local development works
# without manually setting environment variables before every command.
# In production Lambda, environment variables are set by the Lambda service
# and .env does not exist; load_dotenv() silently does nothing.
python-dotenv==1.1.0
